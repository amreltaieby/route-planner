"use client";
import { Fragment, useEffect, useState } from "react";
import { supabase } from "../db";

const DAYS = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
const toArr = (s) =>
  (s || "")
    .split(",")
    .map((x) => x.trim())
    .filter(Boolean);
const arr = (x) => (Array.isArray(x) ? x : []);
const digits = (s) => (s || "").replace(/\D/g, "").length;

export default function Clients() {
  const [rows, setRows] = useState([]);
  const [trips, setTrips] = useState([]);
  const [locs, setLocs] = useState([]);
  const [dests, setDests] = useState([]);
  const [users, setUsers] = useState({});
  const [err, setErr] = useState("");
  const [menu, setMenu] = useState(null);
  const [editId, setEditId] = useState(null);
  const [ed, setEd] = useState(null);
  const [saving, setSaving] = useState(false);
  const [editErr, setEditErr] = useState("");
  const [askDel, setAskDel] = useState(null);
  const [deleting, setDeleting] = useState(false);

  useEffect(() => {
    async function init() {
      const [c, t, l, d, u] = await Promise.all([
        supabase
          .from("clients")
          .select("*")
          .eq("status", "approved")
          .order("created_at", { ascending: true }),
        supabase.from("trips").select("id,name,data"),
        supabase.from("locations").select("name").order("name"),
        supabase.from("destinations").select("name").order("name"),
        supabase.rpc("operator_client_usernames"),
      ]);
      if (c.error) setErr(c.error.message);
      else setRows(c.data || []);
      setTrips((t.data || []).filter((x) => x.data && x.data.day));
      setLocs((l.data || []).map((x) => x.name));
      setDests((d.data || []).map((x) => x.name));
      const m = {};
      (u.data || []).forEach((x) => {
        m[x.client_id] = x.username;
      });
      setUsers(m);
    }
    init();
  }, []);

  const tripName = (id) => {
    const t = trips.find((x) => x.id === id);
    return t ? t.name : null;
  };

  function beginEdit(r) {
    const parts = (r.name || "").split(" ");
    setEditErr("");
    setMenu(null);
    setEditId(r.id);
    setEd({
      first: r.first_name || (r.last_name ? "" : parts[0] || ""),
      last: r.last_name || (r.first_name ? "" : parts.slice(1).join(" ")),
      phone: r.phone || "",
      parent: r.parent_phone || "",
      stop: r.stop || "",
      destination: r.destination || "",
      days: DAYS.filter((d) => toArr(r.days).includes(d)),
    });
  }

  async function saveEdit(r) {
    setEditErr("");
    if (!ed.first.trim() || !ed.last.trim()) return setEditErr("First and last name are required.");
    if (digits(ed.phone) < 7) return setEditErr("Enter a valid phone number.");
    if (ed.parent.trim() && digits(ed.parent) < 7)
      return setEditErr("Enter a valid parent phone number or leave it empty.");
    const ch = {};
    if (ed.first.trim() !== (r.first_name || "")) ch.first_name = ed.first.trim();
    if (ed.last.trim() !== (r.last_name || "")) ch.last_name = ed.last.trim();
    if (ed.phone.trim() !== (r.phone || "")) ch.phone = ed.phone.trim();
    if (ed.parent.trim() !== (r.parent_phone || "")) ch.parent_phone = ed.parent.trim();
    if (ed.stop !== (r.stop || "")) ch.stop = ed.stop;
    if (ed.destination !== (r.destination || "")) ch.destination = ed.destination;
    const nd = DAYS.filter((d) => ed.days.includes(d)).join(", ");
    const od = DAYS.filter((d) => toArr(r.days).includes(d)).join(", ");
    if (nd !== od) ch.days = nd;
    if (Object.keys(ch).length === 0) {
      setEditId(null);
      return;
    }
    setSaving(true);
    const res = await supabase.rpc("operator_propose_client_edit", {
      p_client_id: String(r.id),
      p_changes: ch,
    });
    setSaving(false);
    if (res.error) return setEditErr(res.error.message);
    setRows((rs) =>
      rs.map((x) =>
        x.id === r.id ? { ...x, pending_edit: { changes: ch }, edit_result: null } : x
      )
    );
    setEditId(null);
  }

  async function doDelete() {
    const r = askDel;
    setDeleting(true);
    const res = await supabase.rpc("operator_delete_client", { p_client_id: String(r.id) });
    setDeleting(false);
    setAskDel(null);
    if (res.error) return setErr(res.error.message);
    setRows((rs) => rs.filter((x) => x.id !== r.id));
  }

  const optsFor = (list, cur) => {
    const l = cur && !list.includes(cur) ? [cur, ...list] : list;
    return l.map((n) => (
      <option key={n} value={n}>
        {n}
      </option>
    ));
  };

  return (
    <main className="rt wide">
      <h1>Clients</h1>
      <p className="muted">
        Edits you make here are not saved right away. The client sees them when they next log in
        and they are applied only if the client agrees.
      </p>
      {err && <p className="err">{err}</p>}

      <table>
        <thead>
          <tr>
            <th>Person name</th>
            <th>Username</th>
            <th>Their stop</th>
            <th>Their destination</th>
            <th>Days</th>
            <th>Details</th>
            <th>ID</th>
            <th></th>
          </tr>
        </thead>
        <tbody>
          {rows.map((r) => {
            const editing = editId === r.id;
            const sel = DAYS.filter((d) => toArr(r.days).includes(d));
            const tb =
              r.trips_by_day && typeof r.trips_by_day === "object" && !Array.isArray(r.trips_by_day)
                ? r.trips_by_day
                : {};
            const enr = sel
              .map((d) => ({
                d,
                names: arr(tb[d]).map(tripName).filter(Boolean),
              }))
              .filter((x) => x.names.length);
            const hasTrips = enr.length > 0 && !editing;
            const nb = hasTrips ? { borderBottom: "none" } : undefined;
            return (
              <Fragment key={r.id}>
                <tr style={nb}>
                  <td style={nb}>
                    {editing ? (
                      <>
                        <input
                          className="in"
                          placeholder="First name"
                          value={ed.first}
                          onChange={(e) => setEd({ ...ed, first: e.target.value })}
                        />
                        <input
                          className="in"
                          style={{ marginTop: 6 }}
                          placeholder="Last name"
                          value={ed.last}
                          onChange={(e) => setEd({ ...ed, last: e.target.value })}
                        />
                      </>
                    ) : (
                      <>
                        <div>{r.name}</div>
                        {r.pending_edit && (
                          <div className="muted">Waiting for client approval</div>
                        )}
                        {!r.pending_edit && r.edit_result === "approved" && (
                          <div className="muted">Client approved the last edit</div>
                        )}
                        {!r.pending_edit && r.edit_result === "declined" && (
                          <div className="muted">Client declined the last edit</div>
                        )}
                      </>
                    )}
                  </td>
                  <td style={nb}>{users[String(r.id)] || "-"}</td>
                  <td style={nb}>
                    {editing ? (
                      <select
                        className="in"
                        value={ed.stop}
                        onChange={(e) => setEd({ ...ed, stop: e.target.value })}
                      >
                        <option value="">No stop</option>
                        {optsFor(locs, ed.stop)}
                      </select>
                    ) : (
                      r.stop || "-"
                    )}
                  </td>
                  <td style={nb}>
                    {editing ? (
                      <select
                        className="in"
                        value={ed.destination}
                        onChange={(e) => setEd({ ...ed, destination: e.target.value })}
                      >
                        <option value="">No destination</option>
                        {optsFor(dests, ed.destination)}
                      </select>
                    ) : (
                      r.destination || "-"
                    )}
                  </td>
                  <td style={nb}>
                    {editing ? (
                      <div>
                        {DAYS.map((d) => (
                          <label
                            key={d}
                            style={{
                              display: "flex",
                              alignItems: "center",
                              gap: 6,
                              margin: "2px 0",
                              cursor: "pointer",
                            }}
                          >
                            <input
                              type="checkbox"
                              checked={ed.days.includes(d)}
                              onChange={() =>
                                setEd({
                                  ...ed,
                                  days: ed.days.includes(d)
                                    ? ed.days.filter((x) => x !== d)
                                    : [...ed.days, d],
                                })
                              }
                            />
                            {d}
                          </label>
                        ))}
                      </div>
                    ) : sel.length ? (
                      sel.join(", ")
                    ) : (
                      "-"
                    )}
                  </td>
                  <td style={nb}>
                    {editing ? (
                      <>
                        <input
                          className="in"
                          type="tel"
                          placeholder="Phone"
                          value={ed.phone}
                          onChange={(e) => setEd({ ...ed, phone: e.target.value })}
                        />
                        <input
                          className="in"
                          style={{ marginTop: 6 }}
                          type="tel"
                          placeholder="Parent phone"
                          value={ed.parent}
                          onChange={(e) => setEd({ ...ed, parent: e.target.value })}

                      <>
                        <div className="muted">Phone: {r.phone || "-"}</div>

                                borderRadius: 8,
                                overflow: "hidden",

                            >
                              <div
                                className="mi"
                                style={{ cursor: "pointer", padding: "10px 14px" }}

                              >
                                Edit entries
                              </div>
                              <div

                            className="muted"

                Cancel
              </button>
            </div>
          </div>
        </div>
      )}
    </main>
  );
}

"use client";
import { useEffect, useState } from "react";
import { supabase } from "../db";

const COORDS = /^-?\d{1,3}(\.\d+)?,-?\d{1,3}(\.\d+)?$/;

export default function Requests() {
  const [rows, setRows] = useState([]);
  const [loaded, setLoaded] = useState(false);
  const [err, setErr] = useState("");
  const [busyId, setBusyId] = useState(null);

  async function load() {
    const { data, error } = await supabase
      .from("clients")
      .select("id,name,phone,parent_phone,stop,stop_coords,destination,created_at")
      .eq("status", "pending")
      .order("created_at", { ascending: true });
    if (error) setErr(error.message);
    setRows(data || []);
    setLoaded(true);
  }

  useEffect(() => {
    load();
  }, []);

  async function decide(r, status) {
    if (status === "rejected" && !window.confirm(`Reject ${r.name}'s request?`)) return;
    setErr("");
    setBusyId(r.id);
    const u = await supabase.from("clients").update({ status }).eq("id", r.id).select();
    setBusyId(null);
    if (u.error) return setErr(u.error.message);
    if (!u.data || !u.data.length)
      return setErr("Update was blocked. Check the Supabase policies.");
    setRows((rs) => rs.filter((x) => x.id !== r.id));
  }

  function stopCell(r) {
    if (!r.stop) return "-";
    const c = (r.stop_coords || "").trim();
    if (c && COORDS.test(c)) {
      return (
        <a
          className="maps"
          href={"https://www.google.com/maps/search/?api=1&query=" + encodeURIComponent(c)}
          target="_blank"
          rel="noopener noreferrer"
        >
          {r.stop}
        </a>
      );
    }
    return r.stop;
  }

  return (
    <main className="rt wide">
      <h1>Client requests</h1>
      <p className="muted">Clients who signed up with your company code and are waiting for approval.</p>
      {err && <p className="err">{err}</p>}

      <table>
        <thead>
          <tr>
            <th>Name</th>
            <th>Phone</th>
            <th>Parent phone</th>
            <th>Stop</th>
            <th>Destination</th>
            <th>Requested</th>
            <th></th>
          </tr>
        </thead>
        <tbody>
          {rows.map((r) => (
            <tr key={r.id}>
              <td>{r.name}</td>
              <td>{r.phone || "-"}</td>
              <td>{r.parent_phone || "-"}</td>
              <td>{stopCell(r)}</td>
              <td>{r.destination || "-"}</td>
              <td>{new Date(r.created_at).toLocaleString()}</td>
              <td>
                <div style={{ display: "flex", gap: 6 }}>
                  <button
                    className="btn"
                    style={{ marginTop: 0 }}
                    disabled={busyId === r.id}
                    onClick={() => decide(r, "approved")}
                  >
                    Approve
                  </button>
                  <button
                    className="ghost"
                    style={{ marginTop: 0 }}
                    disabled={busyId === r.id}
                    onClick={() => decide(r, "rejected")}
                  >
                    Reject
                  </button>
                </div>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
      {loaded && rows.length === 0 && <div className="empty">No pending requests.</div>}
    </main>
  );
}

const fmt = (ms) =>

const fmtDur = (min) => {
  const t = Math.round(min);
  const h = Math.floor(t / 60);
  const m = t % 60;
  return h ? `${h} h ${m} min` : `${m} min`;
};

const parseKm = (s) => {
  const t = String(s).trim().toLowerCase().replace(",", ".");
  const m = t.match(/^(\d+(?:\.\d+)?)\s*(km|m)?$/);
  if (!m) return NaN;
  const v = +m[1];
  return m[2] === "m" ? v / 1000 : v;
};

const parseMin = (s) => {
  const t = String(s).trim().toLowerCase();
  if (!t) return NaN;
  let m = t.match(/^(\d+(?:\.\d+)?)$/);
  if (m) return +m[1];
  m = t.match(/^(\d+):(\d{1,2})$/);
  if (m) return +m[1] * 60 + +m[2];
  const h = t.match(/(\d+(?:\.\d+)?)\s*h/);
  const mi = t.match(/(\d+(?:\.\d+)?)\s*m/);
  if (h || mi) return (h ? +h[1] * 60 : 0) + (mi ? +mi[1] : 0);
  return NaN;
};

const legUrl = (a, b) =>
  "https://www.google.com/maps/dir/?api=1&travelmode=driving" +
  `&origin=${a.lat},${a.lng}&destination=${b.lat},${b.lng}`;

function build(r, gMin) {
  const gSec = gMin * 60;
  const raw = r.legSec.reduce((a, b) => a + b, 0);
  const f = gSec / raw;
  const mid = r.order.length - 2;
  const totalSec = gSec + (mid * r.stopMs) / 1000;
  const startMs = r.arrivalMs - totalSec * 1000;
  let t = startMs;
  const order = [{ ...r.order[0], arriveMs: t, departMs: t }];
  for (let i = 1; i < r.order.length; i++) {
    t += r.legSec[i - 1] * f * 1000;
    const arriveMs = t;
    if (i < r.order.length - 1) t += r.stopMs;
    order.push({ ...r.order[i], arriveMs, departMs: t });
  }
  return { factor: f, startMs, order, driveMin: gMin, totalMin: totalSec / 60 };
}

      )}
    </>

  async function saveTrip() {

    if (!tripName.trim()) return setSaveErr("Enter a trip name.");

      return setSaveErr("Time and distance cannot be negative.");

      ...res.plan,
      traffic: fin ? Number(fin.factor.toFixed(3)) : 1,

      .from("trips")
      .insert({ name: tripName.trim(), time_min: t, distance_km: d, data });

      ))}

  );

"use client";
import { useEffect, useState } from "react";
import { usePathname, useRouter } from "next/navigation";
import { supabase } from "./db";

const PUBLIC = ["/login", "/signup", "/join", "/terms"];
const AUTHPAGES = ["/login", "/signup", "/join"];
const isIn = (path, list) => list.some((p) => path === p || path.startsWith(p + "/"));
const CK = "routiq_role_";

export default function AuthGate({ children }) {
  const path = usePathname();
  const router = useRouter();
  const [role, setRole] = useState(undefined);

  useEffect(() => {
    let alive = true;

    async function resolve(session) {
      if (!session) {
        if (alive) setRole(null);
        return;
      }
      const key = CK + session.user.id;
      let cached = null;
      try {
        cached = localStorage.getItem(key);
      } catch {}
      if (cached && alive) setRole(cached);
      const { data } = await supabase
        .from("profiles")
        .select("role")
        .eq("user_id", session.user.id)
        .maybeSingle();
      const r = data ? data.role : null;
      try {
        if (r) localStorage.setItem(key, r);
        else localStorage.removeItem(key);
      } catch {}
      if (alive) setRole(r);
    }

    supabase.auth.getSession().then(({ data }) => resolve(data.session));
    const { data: sub } = supabase.auth.onAuthStateChange((evt, session) => {
      if (evt === "INITIAL_SESSION" || evt === "TOKEN_REFRESHED") return;
      setTimeout(() => resolve(session), 0);
    });
    return () => {
      alive = false;
      sub.subscription.unsubscribe();
    };
  }, []);

  const pub = isIn(path, PUBLIC);
  const clientArea = isIn(path, ["/client"]);
  let target = null;
  if (role !== undefined) {
    if (role === null && !pub) target = "/login";
    else if (role === "operator" && (clientArea || isIn(path, AUTHPAGES))) target = "/";
    else if (role === "client" && !(clientArea || isIn(path, ["/terms"]))) target = "/client";
  }

  useEffect(() => {
    if (target) router.replace(target);
  }, [target]);

  if (target) return null;

  return (
    <div style={{ display: "contents", visibility: role === undefined ? "hidden" : "visible" }}>
      {children}
    </div>
  );
}
