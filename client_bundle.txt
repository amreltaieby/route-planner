"use client";
import { useEffect, useState } from "react";
import Link from "next/link";
import { supabase } from "../db";

const DAYS = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
const MAXT = 2;
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const toArr = (s) =>
  (s || "")
    .split(",")
    .map((x) => x.trim())
    .filter(Boolean);
const typeOf = (t) => (t && t.data && t.data.type === "return" ? "return" : "arrival");
const tword = (ty) => (ty === "return" ? "Return" : "Arrival");
const placeOf = (t) => {
  const p = typeOf(t) === "return" ? t.data.start : t.data.end;
  return (p && p.name) || t.name;
};
const norm = (s) => (s || "").trim().toLowerCase();
const LABELS = {
  first_name: "First name",
  last_name: "Last name",
  phone: "Phone",
  parent_phone: "Parent phone",
  stop: "Stop",
  destination: "Destination",
  days: "Days",
};
const showVal = (k, v) => {
  if (k === "days") {
    const l = DAYS.filter((d) => toArr(v).includes(d));
    return l.length ? l.join(", ") : "-";
  }
  return v ? String(v) : "-";
};

function lockedNow(tz) {
  try {
    const parts = new Intl.DateTimeFormat("en-US", {
      timeZone: tz,
      weekday: "long",
      hour: "numeric",
      hourCycle: "h23",
    }).formatToParts(new Date());
    const wd = parts.find((p) => p.type === "weekday").value;
    const hr = Number(parts.find((p) => p.type === "hour").value);
    const out = [wd];
    if (hr >= 21) out.push(DAYS[(DAYS.indexOf(wd) + 1) % 7]);
    return out;
  } catch {
    return [];
  }
}

export default function ClientHome() {
  const [me, setMe] = useState(null);
  const [noSession, setNoSession] = useState(false);
  const [trips, setTrips] = useState([]);
  const [tz, setTz] = useState("Africa/Cairo");
  const [sel, setSel] = useState([]);
  const [joined, setJoined] = useState({});
  const [ready, setReady] = useState(false);
  const [err, setErr] = useState("");
  const [msg, setMsg] = useState("");
  const [saving, setSaving] = useState(false);
  const [ask, setAsk] = useState(null);
  const [note, setNote] = useState("");
  const [answering, setAnswering] = useState(false);
  const [editErr, setEditErr] = useState("");
  const [, setTick] = useState(0);

  useEffect(() => {
    const i = setInterval(() => setTick((x) => x + 1), 60000);
    return () => clearInterval(i);
  }, []);

  useEffect(() => {
    async function fetchMe(uid) {
      return supabase
        .from("clients")
        .select(
          "id,first_name,last_name,phone,parent_phone,stop,status,days,trips_by_day,destination,pending_edit"
        )
        .eq("user_id", uid)
        .maybeSingle();
    }

    async function init() {
      const s = await supabase.auth.getSession();
      const uid = s.data.session && s.data.session.user.id;
      if (!uid) {
        setNoSession(true);
        setReady(true);
        return;
      }
      let c = await fetchMe(uid);
      if (!c.data) {
        await sleep(800);
        c = await fetchMe(uid);
      }
      if (c.data) {
        setMe(c.data);
        setSel(DAYS.filter((d) => toArr(c.data.days).includes(d)));
        const tb =
          c.data.trips_by_day &&
          typeof c.data.trips_by_day === "object" &&
          !Array.isArray(c.data.trips_by_day)
            ? c.data.trips_by_day
            : {};
        const j = {};
        Object.keys(tb).forEach((k) => {
          if (Array.isArray(tb[k])) j[k] = tb[k];
        });
        setJoined(j);
        const [t, co] = await Promise.all([
          supabase.from("trips").select("id,name,data"),
          supabase.from("companies").select("tz").maybeSingle(),
        ]);
        setTrips((t.data || []).filter((x) => x.data && x.data.day));
        if (co.data && co.data.tz) setTz(co.data.tz);
      }
      setReady(true);
    }
    init();
  }, []);

  const locked = lockedNow(tz);

  const myDest = norm(me && me.destination);
  const tripsFor = (d) =>
    trips
      .filter(
        (t) =>
          t.data.day === d &&
          (!myDest || norm(placeOf(t)) === myDest || (joined[d] || []).includes(t.id))
      )
      .sort((a, b) => (a.data.arrival || "99:99").localeCompare(b.data.arrival || "99:99"));

  const sameTypeJoined = (ids, ty) =>
    ids.some((id) => {
      const o = trips.find((x) => x.id === id);
      return o && typeOf(o) === ty;
    });

  async function apply(nextSel, nextJoined) {
    const prev = { sel, joined };
    setErr("");
    setMsg("");
    setSel(nextSel);
    setJoined(nextJoined);
    setSaving(true);
    const tb = {};
    nextSel.forEach((d) => {
      const ids = nextJoined[d] || [];
      if (ids.length) tb[d] = ids;
    });
    const r = await supabase.rpc("book_trips", {
      p_days: DAYS.filter((d) => nextSel.includes(d)).join(", "),
      p_trips: tb,
    });
    setSaving(false);
    if (r.error) {
      setSel(prev.sel);
      setJoined(prev.joined);
      setErr(r.error.message);
      return;
    }
    setMsg("Saved.");
  }

  function pickDay(d) {
    if (saving || locked.includes(d)) return;
    if (sel.includes(d)) {
      const n = (joined[d] || []).length;
      setAsk({
        text: n
          ? `Unselect ${d}? This will also remove your ${n} trip${n > 1 ? "s" : ""} on that day.`
          : `Unselect ${d}?`,
        yes: () => {
          const nj = { ...joined };
          delete nj[d];
          apply(
            sel.filter((x) => x !== d),
            nj
          );
        },
      });
    } else apply([...sel, d], joined);
  }

  function pickTrip(d, t) {
    if (saving || locked.includes(d)) return;
    const list = joined[d] || [];
    const ty = typeOf(t);
    if (list.includes(t.id)) {
      setAsk({
        text: `Leave "${placeOf(t)} | ${tword(ty)}" on ${d}?`,
        yes: () => apply(sel, { ...joined, [d]: list.filter((x) => x !== t.id) }),
      });
    } else if (sameTypeJoined(list, ty)) {
      setNote(
        `You already joined a ${ty} trip on ${d}. Joining another one would exceed the limit of one arrival trip and one return trip per day. To switch, leave your current ${ty} trip first.`
      );
    } else apply(sel, { ...joined, [d]: [...list, t.id] });
  }

  async function respond(ok) {
    setEditErr("");
    setAnswering(true);
    const r = await supabase.rpc("client_respond_edit", { p_approve: ok });
    if (r.error) {
      setAnswering(false);
      setEditErr(r.error.message);
      return;
    }
    window.location.reload();
  }

  async function logout() {
    await supabase.auth.signOut();
    window.location.assign("/login");
  }

  if (!ready) return null;

  const approved = me && me.status === "approved";
  const changes =
    approved && me.pending_edit && me.pending_edit.changes ? me.pending_edit.changes : null;
  const changeKeys = changes ? Object.keys(changes).filter((k) => LABELS[k]) : [];

  return (
    <main className="rt" style={{ maxWidth: 520 }}>
      <h1 className="brandh">Routiq</h1>

      {noSession && (
        <p className="muted" style={{ textAlign: "center" }}>
          You are not logged in.{" "}
          <Link className="maps" href="/login">
            Log in
          </Link>
        </p>
      )}

      {!noSession && !me && (
        <section className="card" style={{ marginTop: 0 }}>
          <div>
            Your client profile was not found. If you just signed up, try again in a moment.
            Otherwise contact your company.
          </div>
          <div className="mbtns">
            <button className="btn" onClick={() => window.location.reload()}>
              Try again
            </button>
            <button className="ghost" onClick={logout}>
              Log out
            </button>
          </div>
        </section>
      )}

      {me && (
        <div
          style={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            gap: 12,
          }}
        >
          <div>
            Welcome, <span className="acc">{me.first_name}</span>
          </div>
          <button className="ghost" style={{ marginTop: 0 }} onClick={logout}>
            Log out
          </button>
        </div>
      )}

      {me && me.status === "pending" && (
        <section className="card">
          <div className="sec" style={{ margin: "0 0 8px" }}>
            Request sent
          </div>
          <div>
            The last step is for your company to approve your join request. You will be able to
            book trips once it is approved.
          </div>
        </section>
      )}

      {me && me.status === "rejected" && (
        <section className="card">
          <div>Your join request was declined. Please contact your company.</div>
        </section>
      )}

      {approved && (
        <>
          <div className="sec" style={{ marginTop: 28 }}>
            Book trips
          </div>
          <p className="muted" style={{ margin: "0 0 4px" }}>
            book your trip early, booking a trip closes at 9pm on the day before the trip
          </p>

          <label>1. Pick your days</label>
          <div className="daygrid">
            {DAYS.map((d) => {
              const lk = locked.includes(d);
              return (
                <button
                  key={d}
                  type="button"
                  className={sel.includes(d) ? "daybtn on" : "daybtn"}
                  style={lk ? { opacity: 0.5, cursor: "default" } : undefined}
                  disabled={saving || lk}
                  onClick={() => pickDay(d)}
                >
                  {d}
                  {lk ? " (closed)" : ""}
                </button>
              );
            })}
          </div>

          <label>2. Join trips (one arrival and one return per day)</label>
          {sel.length === 0 && <div className="muted">Pick a day to see its trips.</div>}
          {DAYS.filter((d) => sel.includes(d)).map((d) => {
            const list = tripsFor(d);
            const mine = joined[d] || [];
            const lk = locked.includes(d);
            return (
              <div key={d} style={{ marginBottom: 14 }}>
                <div className="sec" style={{ margin: "14px 0 0", fontSize: 15 }}>
                  {d}{" "}
                  <span className="muted">
                    ({mine.length}/{MAXT} trips)
                  </span>
                </div>
                {lk && <div className="muted">Booking for this day is closed.</div>}
                {list.length === 0 && <div className="muted">No trips available.</div>}
                {list.map((t) => {
                  const ty = typeOf(t);
                  const on = mine.includes(t.id);
                  const full = !on && sameTypeJoined(mine, ty);
                  return (
                    <div className="tripr" key={t.id}>
                      <div>
                        <div>{placeOf(t) + " | " + tword(ty)}</div>
                        <div className="muted">
                          {t.data.arrival ? `${tword(ty)} ${t.data.arrival}` : "No time set"}
                        </div>
                      </div>
                      <button
                        className={on ? "btn" : "ghost"}
                        style={{ marginTop: 0, ...(full ? { opacity: 0.5 } : {}) }}
                        disabled={saving || lk}
                        onClick={() => pickTrip(d, t)}
                      >
                        {on ? "Joined" : "Join"}
                      </button>
                    </div>
                  );
                })}
              </div>
            );
          })}

          <div style={{ minHeight: 22, marginTop: 8 }}>
            {saving && <span className="muted">Saving...</span>}
            {!saving && msg && <span className="muted">{msg}</span>}
          </div>
          {err && <p className="err">{err}</p>}
        </>
      )}

      <p className="muted" style={{ marginTop: 28 }}>
        <Link className="maps" href="/terms">
          Terms and policy
        </Link>
      </p>

      {ask && (
        <div className="ov" onClick={() => setAsk(null)}>
          <div className="modal" onClick={(e) => e.stopPropagation()}>
            <h3>Are you sure?</h3>
            <p className="muted" style={{ marginTop: 10 }}>
              {ask.text}
            </p>
            <div className="mbtns">
              <button
                className="btn"
                onClick={() => {
                  const y = ask.yes;
                  setAsk(null);
                  y();
                }}
              >
                Yes
              </button>
              <button className="ghost" onClick={() => setAsk(null)}>
                Cancel
              </button>
            </div>
          </div>
        </div>
      )}

      {note && (
        <div className="ov" onClick={() => setNote("")}>
          <div className="modal" onClick={(e) => e.stopPropagation()}>
            <h3>Trip limit reached</h3>
            <p className="muted" style={{ marginTop: 10 }}>
              {note}
            </p>
            <div className="mbtns">
              <button className="btn" onClick={() => setNote("")}>
                OK
              </button>
            </div>
          </div>
        </div>
      )}

      {changes && changeKeys.length > 0 && (
        <div className="ov">
          <div className="modal" onClick={(e) => e.stopPropagation()}>
            <h3>Your company wants to change your details</h3>
            <p className="muted" style={{ marginTop: 10 }}>
              Nothing is changed until you agree. Please check each change:
            </p>
            <div style={{ marginTop: 10 }}>
              {changeKeys.map((k) => (
                <div key={k} style={{ margin: "8px 0" }}>
                  <div className="muted">{LABELS[k]}</div>
                  <div>
                    {showVal(k, me[k]) + " \u2192 "}
                    <b>{showVal(k, changes[k])}</b>
                  </div>
                </div>
              ))}
            </div>
            {changeKeys.includes("days") && (
              <p className="muted">
                Days you no longer have will also remove your trips on those days.
              </p>
            )}
            {editErr && <p className="err">{editErr}</p>}
            <div className="mbtns">
              <button className="btn" onClick={() => respond(true)} disabled={answering}>
                {answering ? "Saving..." : "Agree"}
              </button>
              <button className="ghost" onClick={() => respond(false)} disabled={answering}>
                Decline
              </button>
            </div>
          </div>
        </div>
      )}
    </main>
  );
}

"use client";
import { useEffect, useState } from "react";
import { usePathname, useRouter } from "next/navigation";
import { supabase } from "./db";

const PUBLIC = ["/login", "/signup", "/join", "/terms"];
const AUTHPAGES = ["/login", "/signup", "/join"];
const isIn = (path, list) => list.some((p) => path === p || path.startsWith(p + "/"));
const CK = "routiq_role_";

export default function AuthGate({ children }) {
  const path = usePathname();
  const router = useRouter();
  const [role, setRole] = useState(undefined);

  useEffect(() => {
    let alive = true;

    async function resolve(session) {
      if (!session) {
        if (alive) setRole(null);
        return;
      }
      const key = CK + session.user.id;
      let cached = null;
      try {
        cached = localStorage.getItem(key);
      } catch {}
      if (cached && alive) setRole(cached);
      const { data } = await supabase
        .from("profiles")
        .select("role")
        .eq("user_id", session.user.id)
        .maybeSingle();
      const r = data ? data.role : null;
      try {
        if (r) localStorage.setItem(key, r);
        else localStorage.removeItem(key);
      } catch {}
      if (alive) setRole(r);
    }

    supabase.auth.getSession().then(({ data }) => resolve(data.session));
    const { data: sub } = supabase.auth.onAuthStateChange((evt, session) => {
      if (evt === "INITIAL_SESSION" || evt === "TOKEN_REFRESHED") return;
      setTimeout(() => resolve(session), 0);
    });
    return () => {
      alive = false;
      sub.subscription.unsubscribe();
    };
  }, []);

  const pub = isIn(path, PUBLIC);
  const clientArea = isIn(path, ["/client"]);
  let target = null;
  if (role !== undefined) {
    if (role === null && !pub) target = "/login";
    else if (role === "operator" && (clientArea || isIn(path, AUTHPAGES))) target = "/";
    else if (role === "client" && !(clientArea || isIn(path, ["/terms"]))) target = "/client";
  }

  useEffect(() => {
    if (target) router.replace(target);
  }, [target]);

  if (target) return null;

  return (
    <div style={{ display: "contents", visibility: role === undefined ? "hidden" : "visible" }}>
      {children}
    </div>
  );
}

  const [copied, setCopied] = useState(false);
  const [eu, setEu] = useState("");
  const [ep, setEp] = useState("");
  const [msg, setMsg] = useState("");
  const [err, setErr] = useState("");
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    try {
      const t = localStorage.getItem("routiq_theme");
      if (t === "light" || t === "dark") setTheme(t);
    } catch {}
    supabase
      .from("companies")
      .select("name,code")
      .maybeSingle()
      .then((r) => setCompany(r.data || null));
  }, []);

  function toggleTheme() {
    const next = theme === "dark" ? "light" : "dark";
    setTheme(next);
    try {
      localStorage.setItem("routiq_theme", next);
    } catch {}
    document.documentElement.setAttribute("data-theme", next);
  }

  const joinLink = company ? `${window.location.origin}/join?c=${company.code}` : "";

  async function copyLink() {
    try {
      await navigator.clipboard.writeText(joinLink);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {}
  }

  async function addEmployee() {
    setErr("");
    setMsg("");
    if (!eu.trim() || !ep) {
      setErr("Enter a username and a password.");
      return;
    }
    setSaving(true);
    try {
      const s = await supabase.auth.getSession();
      const token = s.data.session ? s.data.session.access_token : "";
      const r = await fetch("/api/employees", {
        method: "POST",
        headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
        body: JSON.stringify({ username: eu, password: ep }),
      });
      const d = await r.json();
      if (d.error) setErr(d.error);
      else {
        setEu("");
        setEp("");
        setMsg("Employee saved.");
      }
    } catch (e) {
      setErr(e.message);
    }
    setSaving(false);
  }

  const dark = theme === "dark";

function PinMap({ value, onPick }) {
  const box = useRef(null);
  const st = useRef({});
  const pick = useRef(onPick);
  const init = useRef(value);
  pick.current = onPick;

  useEffect(() => {
    let dead = false;
    (async () => {
      const mod = await import("leaflet");
      const L = mod.default || mod;
      if (dead || !box.current) return;
      const map = L.map(box.current).setView([30.0444, 31.2357], 11);
      L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
        maxZoom: 19,
        attribution: "&copy; OpenStreetMap contributors",
      }).addTo(map);
      st.current = { map, L, marker: null };
      const place = (ll) => {
        const s = st.current;
        if (s.marker) s.marker.setLatLng(ll);
        else
          s.marker = s.L.circleMarker(ll, {
            radius: 9,
            color: "#ffffff",
            weight: 2,
            fillColor: "#22d3ee",
            fillOpacity: 1,
          }).addTo(s.map);
      };
      if (init.current) {
        const p = init.current.split(",").map(Number);
        if (p.length === 2 && p.every(Number.isFinite)) {
          place(L.latLng(p[0], p[1]));
          map.setView([p[0], p[1]], 16);
        }
      }
      map.on("click", (e) => {
        place(e.latlng);
        pick.current(e.latlng.lat.toFixed(6) + "," + e.latlng.lng.toFixed(6));
      });
    })();
    return () => {
      dead = true;
      if (st.current.map) st.current.map.remove();
      st.current = {};
    };
  }, []);

  useEffect(() => {
    const s = st.current;
    if (!value && s.marker) {
      s.marker.remove();
      s.marker = null;
    }
  }, [value]);

  return (
    <div
      ref={box}
      style={{
        height: 260,
        borderRadius: 8,
        border: "1px solid var(--border)",
        position: "relative",
        zIndex: 0,
        isolation: "isolate",
      }}
    />
  );
}

  const label = mode === "map" ? "Choose on map" : value || "Select your stop";

  function close() {
    setOpen(false);
    setQ("");
  }

  return (
    <div className="pk">
      <button type="button" className="in cp" disabled={disabled} onClick={() => setOpen(!open)}>
        {disabled ? "Enter your company code first" : label}
      </button>
      {open && (
        <>
          <div className="mo" onClick={close} />
          <div
            className="mlist"
            style={{
              width: "100%",
              display: "flex",
              flexDirection: "column",
              maxHeight: 300,
              overflow: "hidden",
            }}
          >
            <div style={{ padding: 8, borderBottom: "1px solid var(--border)" }}>
              <input
                className="in"
                placeholder="Search for your stop"
                value={q}
                onChange={(e) => setQ(e.target.value)}
              />
            </div>
            <div style={{ overflow: "auto" }}>
              <div
                className="mi"
                style={{ color: "var(--cyan)", fontWeight: 600 }}
                onClick={() => {
                  onMap();
                  close();
                }}
              >
                Choose on map
              </div>
              {list.map((s) => (
                <div
                  className="mi"
                  key={s}
                  onClick={() => {
                    onList(s);
                    close();
                  }}
                >
                  {s}
                </div>
              ))}
              {list.length === 0 && <div className="mi muted">No matches</div>}
            </div>
          </div>
        </>
      )}
    </div>
  );
}

export default function Join() {
  const [f, setF] = useState({
    code: "",
    username: "",
    email: "",
    password: "",
    first: "",
    last: "",
    phone: "",
    parentPhone: "",
    stop: "",
    pinName: "",
    destination: "",
    stopCoords: "",
  });
  const [mode, setMode] = useState("");
  const [stops, setStops] = useState([]);
  const [dests, setDests] = useState([]);
  const [err, setErr] = useState("");
  const [busy, setBusy] = useState(false);
  const set = (k) => (e) => setF((x) => ({ ...x, [k]: e.target.value }));
  const digits = (s) => s.replace(/\D/g, "").length;

  useEffect(() => {
    const c = new URLSearchParams(window.location.search).get("c");
    if (c) setF((x) => ({ ...x, code: c.toUpperCase() }));
  }, []);

  useEffect(() => {
    const code = f.code.trim();
    if (code.length < 4) {
      setStops([]);
      setDests([]);
      return;
    }
    let alive = true;
    supabase.rpc("company_locations", { p_code: code }).then((r) => {
      if (alive) setStops((r.data || []).map((x) => x.name));
    });
    supabase.rpc("company_destinations", { p_code: code }).then((r) => {
      if (alive) setDests((r.data || []).map((x) => x.name));
    });
    return () => {
      alive = false;
    };
  }, [f.code]);

  function onEnter(e) {
    if (e.key !== "Enter") return;
    const t = e.target;
    if (t.tagName === "BUTTON" || t.tagName === "TEXTAREA") return;
    e.preventDefault();
    if (t.closest(".mlist")) return;
    const els = Array.from(e.currentTarget.querySelectorAll("input, select, button.cp")).filter(
      (x) => !x.disabled && !x.closest(".mlist")
    );
    const i = els.indexOf(t);
    if (i === -1) return;
    if (i === els.length - 1) submit();
    else els[i + 1].focus();
  }

  async function submit() {
    setErr("");
    const user = cleanUser(f.username);
    const isMap = mode === "map";
    const stopName = isMap ? f.pinName.trim() : f.stop;
    if (!f.code.trim()) return setErr("Enter your company code.");
    if (!validUser(user))
      return setErr("Username: 3 to 30 characters, letters, numbers, dot, dash or underscore.");
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(f.email.trim()))
      return setErr("Enter a valid email address.");
    if (f.password.length < 8) return setErr("Password must be at least 8 characters.");
    if (!f.first.trim() || !f.last.trim()) return setErr("Enter your first and last name.");
    if (digits(f.phone) < 7) return setErr("Enter a valid phone number.");
    if (f.parentPhone.trim() && digits(f.parentPhone) < 7)
      return setErr("Enter a valid parent phone number or leave it empty.");
    if (isMap) {
      if (!f.stopCoords) return setErr("Tap the map to drop your pin.");
      if (!stopName) return setErr("Name your stop.");
      if (stops.some((s) => s.toLowerCase() === stopName.toLowerCase()))
        return setErr("A stop with that name already exists. Pick it from the list or use a different name.");
    } else if (stops.length > 0 && !f.stop) return setErr("Pick your stop.");
    if (dests.length > 0 && !f.destination) return setErr("Pick your destination.");
    setBusy(true);
    const su = await supabase.auth.signUp({ email: toEmail(user), password: f.password });
    if (su.error) {
      setBusy(false);
      return setErr(
        /already/i.test(su.error.message) ? "That username is taken." : su.error.message
      );
    }
    if (!su.data.session) {
      setBusy(false);
      return setErr("Email confirmation is still on in Supabase. Turn it off and try again.");
    }
    const r = await supabase.rpc("join_company", {
      p_code: f.code.trim(),
      p_username: user,
      p_first: f.first.trim(),
      p_middle: "",
      p_last: f.last.trim(),
      p_phone: f.phone.trim(),
      p_parent_phone: f.parentPhone.trim(),
      p_stop: stopName,
      p_destination: f.destination,
      p_stop_coords: isMap ? f.stopCoords : null,
      p_email: f.email.trim(),
    });
    if (r.error) {
      await supabase.auth.signOut();
      setBusy(false);
      return setErr(r.error.message);
    }
    window.location.assign("/client");
  }

  const destOpts = dests.map((s) => (
    <option key={s} value={s}>
      {s}
    </option>
  ));
  const destHint = dests.length ? "Select your destination" : "Enter a valid company code first";

  return (
    <main className="rt" style={{ maxWidth: 480 }}>
      <h1 className="brandh">Routiq</h1>
      <h2 style={{ textAlign: "center", margin: "0 0 8px" }}>Join your company</h2>
      <form
        noValidate
        onKeyDown={onEnter}
        onSubmit={(e) => {
          e.preventDefault();
          submit();
        }}
      >
        <label>Company code</label>
        <input className="in" value={f.code} onChange={set("code")} />

        <label>Username</label>
        <input className="in" autoComplete="username" value={f.username} onChange={set("username")} />

        <label>Email address</label>
        <input
          className="in"
          type="email"
          autoComplete="email"
          value={f.email}
          onChange={set("email")}
        />

        <label>Password</label>
        <input
          className="in"
          type="password"
          autoComplete="new-password"
          value={f.password}
          onChange={set("password")}
        />

        <div style={{ display: "flex", gap: 10 }}>
          <div style={{ flex: 1, minWidth: 0 }}>
            <label>First name</label>
            <input className="in" value={f.first} onChange={set("first")} />
          </div>
          <div style={{ flex: 1, minWidth: 0 }}>
            <label>Last name</label>
            <input className="in" value={f.last} onChange={set("last")} />
          </div>
        </div>

        <label>Phone number</label>
        <input className="in" type="tel" value={f.phone} onChange={set("phone")} />

        <label>Parent phone number (optional)</label>
        <input className="in" type="tel" value={f.parentPhone} onChange={set("parentPhone")} />

        <label>Your stop</label>
        <StopPicker
          disabled={f.code.trim().length < 4}
          stops={stops}
          mode={mode}
          value={f.stop}
          onList={(s) => {
            setMode("list");
            setF((x) => ({ ...x, stop: s, stopCoords: "", pinName: "" }));
          }}
          onMap={() => {
            setMode("map");
            setF((x) => ({ ...x, stop: "" }));
          }}
        />

        {mode === "map" && (
          <>
            <label>Tap the map to drop a pin on your exact stop</label>
            <PinMap value={f.stopCoords} onPick={(v) => setF((x) => ({ ...x, stopCoords: v }))} />
            <p className="muted" style={{ margin: "6px 0 0" }}>
              {f.stopCoords ? "Pin placed: " + f.stopCoords + " " : "No pin placed yet. "}
              {f.stopCoords && (
                <button
                  type="button"
                  className="ghost"
                  onClick={() => setF((x) => ({ ...x, stopCoords: "" }))}
                >
                  Clear pin
                </button>
              )}
            </p>

            <label>Name this stop</label>
            <input
              className="in"
              placeholder="Lake View Compound Gate 3"
              value={f.pinName}
              onChange={set("pinName")}
            />
            <p className="muted" style={{ margin: "6px 0 0" }}>
              {"Name the place, for example \"Lake View Compound Gate 3\". Do not use a personal name such as \"Doe's house\"."}
            </p>
          </>
        )}

        <label>Your destination</label>
        <select className="in" value={f.destination} onChange={set("destination")}>
          <option value="">{destHint}</option>
          {destOpts}
        </select>

        <button className="btn" type="submit" disabled={busy}>
          {busy ? "Signing up..." : "Sign up"}
        </button>
      </form>
      {err && (
        <p className="err" role="alert">
          {err}
        </p>
      )}
      <p className="muted" style={{ marginTop: 22 }}>
        By signing up you accept the{" "}
        <Link className="maps" href="/terms">
          terms and policy
        </Link>
        .
      </p>
      <p className="muted">
        Already have an account?{" "}
        <Link className="maps" href="/login">
          Log in
        </Link>
      </p>
    </main>
  );
}
